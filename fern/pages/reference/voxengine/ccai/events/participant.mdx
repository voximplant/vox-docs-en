<a id="created" />

</ParamField>

<a id="response" />
<ParamField path="Response" type="'AI.Events.CcaiParticipantResponse'">

</ParamField>

<ParamField path="PlaybackFinished" type="'AI.Events.CcaiParticipantPlaybackFinished'">

<a id="playbackstarted" />
<ParamField path="PlaybackStarted" type="'AI.Events.CcaiParticipantPlaybackStarted'">

</ParamField>

<a id="playbackstopped" />
<ParamField path="PlaybackStopped" type="'AI.Events.CcaiParticipantPlaybackStopped'">
Triggered when playback of a single phrase has stoped.
</ParamField>

<a id="playbackready" />
<ParamField path="PlaybackReady" type="'AI.Events.CcaiParticipantPlaybackReady'">

</ParamField>

<a id="markerreached" />
